SELECT
  member.memberSurname,
  member.memberForename,
  narrow.narrowDesc
FROM member
JOIN memberNarrow ON member.memberId = memberNarrow.memberNarrowMemberId
JOIN narrow ON memberNarrow.memberNarrowNarrowId = narrow.narrowId
JOIN broad ON narrow.narrowBroadId = broad.broadId
WHERE broad.broadDesc = 'Arms Control and Non-Proliferation Studies'
  AND member.memberId NOT IN (
    SELECT memberBroad.memberBroadMemberId
    FROM memberBroad
    JOIN broad AS b2 ON memberBroad.memberBroadBroadId = b2.broadId
    WHERE b2.broadDesc = 'Environmental Security'
  )
ORDER BY member.memberSurname, member.memberForename
